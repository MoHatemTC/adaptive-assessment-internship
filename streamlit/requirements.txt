# Dependencies for the tester harness — the UI's own AND the engine's.
#
# SELF-CONTAINED ON PURPOSE. Streamlit Cloud installs exactly one requirements file: the
# one beside the entry point it was given. It never reads backend/requirements.txt, so an
# earlier version of this file that listed only streamlit and pandas deployed an app that
# died on `ModuleNotFoundError: No module named 'pydantic'` before rendering anything.
#
# The engine constraints below MUST match backend/requirements.txt. `test_streamlit_ui.py`
# asserts they do, so drift fails the suite rather than the next deployment.

# --- the UI itself ---
# We use Streamlit's HTML rendering surface (st.html) in the DAG panel.
streamlit>=1.33,<2
pandas>=2.1,<3

# --- the engine (mirrors backend/requirements.txt) ---
# Config & validation
pydantic>=2.9,<3
pydantic-settings>=2.0,<3

# LLM: all calls route through the LiteLLM proxy
openai>=1.50,<3
httpx>=0.27

# Ability estimation. The posterior is a grid, so this is a hard requirement.
numpy>=1.26,<3

# Untrusted code execution. 2.x only: 1.x has no Sandbox.create() and no files/commands
# API, which the engine needs to run submissions outside the sandbox's notebook kernel.
e2b-code-interpreter>=2.0,<3

# Live interviewer (open/voice answers). Picker + open rubric grader still use LiteLLM.
google-genai>=1.0
websockets>=13
fastapi>=0.110
uvicorn[standard]>=0.27
python-multipart>=0.0.9

# Tracing. Runtime-optional: without LANGFUSE_PUBLIC_KEY and LANGFUSE_SECRET_KEY the app
# runs untraced and otherwise unaffected. Installed anyway so that setting the two keys in
# Streamlit Cloud's secrets is the whole of the work.
langfuse>=3.0,<5
